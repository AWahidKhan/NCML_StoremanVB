--Query For Total Capacity Of Location  
---- Without Date & Location Filter
Select ML.LocationName,Sum(MG.GodownCapacity) as TotalLocationCapacity From Mst_Godown MG 
Inner Join Mst_CMP MC On MG.CMPID = MC.CMPID
Inner Join Mst_Location ML On MC.LocationID = ML.LocationID
--Where ML.LocationName = 'Jaipur'
Group By ML.LocationName


---- With Date & Location Filter (Godown Wise)
Select ML.LocationName,MG.GodownCapacity From Mst_Godown MG 
Inner Join Mst_CMP MC On MG.CMPID = MC.CMPID
Inner Join Mst_Location ML On MC.LocationID = ML.LocationID
Where ((MG.StartDate <= '2009-08-21' and (MG.CloseDate >= '2009-07-30' Or MG.CloseDate Is Null)) And  ML.LocationName = 'Jaipur')

----- With Date & Location Filter (Total of Godown Wise)
Select ML.LocationName,Sum(MG.GodownCapacity) as TotalCapacity From Mst_Godown MG 
Inner Join Mst_CMP MC On MG.CMPID = MC.CMPID
Inner Join Mst_Location ML On MC.LocationID = ML.LocationID
Where ((MG.StartDate <= '2009-12-21' and (MG.CloseDate >= '2008-07-30' Or MG.CloseDate Is Null)) And  ML.LocationName = 'Jaipur')
Group By ML.LocationName

------------Query For Employee List which are in Store-Man but Not in Employee CTC List

---------------Without filter
Select * from Mst_Employee Where EmployeeNo Not In (Select EmployeeCode From Txn_EmpCTC) Order By EmployeeNo

---------------With Designation Filter
Select MD.Designation,ME.EmployeeID,ME.EmployeeNo,ME.EmployeeName from Mst_Employee ME
Inner Join Mst_Designation MD On ME.DesignationID = MD.DesignationID
Where EmployeeNo Not In (Select EmployeeCode From Txn_EmpCTC) And 
(MD.Designation = 'RM' Or MD.Designation = 'AM' Or MD.Designation = 'DRC' Or MD.Designation = 'CMPM' Or MD.Designation = 'RAG')
Order By EmployeeNo
