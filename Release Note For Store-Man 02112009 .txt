Public Function GetTDSDescription(CID_ As Variant) As String
Dim RetVal As String
RetVal = 0
Call TableMst_TDS(" Where TDSID = " & CID_)

   RetVal = RsMst_TDS!Description
End If
GetTDSDescription = RetVal
End Function
